# خطة إضافة المقالات والمنتدى — GAMES ARABIC

---

## الميزتان المراد إضافتهما

### 1. قسم المقالات (مقالات)
مقالات عن ألعاب، أخبار تعريب، أدلّة الترجمة، مراجعات. فيها أقسام، شارات، مقالات مميزة، محتوى Markdown، تعليقات، SEO.

### 2. منتدى المجتمع (منتدى)
نظام نقاشات حيث المستخدمين ي互相following يتناقشوا. أقسام مواضيع، ردود، إعجابات، مشاهدات. متابعة مواضيع للحصول على إشعارات.

```prisma
// ===== المقالات =====
model Article {
  id              String          @id @default(cuid())
  slug            String          @unique
  title           String
  summary         String          @default("")
  content         String          // markdown
  coverUrl        String          @default("")
  authorId        String
  author          User            @relation(fields: [authorId], references: [id], onDelete: Cascade)
  categoryId      String?
  category        ArticleCategory? @relation(fields: [categoryId], references: [id], onDelete: SetNull)
  status          String          @default("draft") // draft | published | archived
  isFeatured      Boolean         @default(false)
  isPinned        Boolean         @default(false)
  views           Int             @default(0)
  likesCount      Int             @default(0)
  commentsCount   Int             @default(0)
  readTime        Int             @default(0)
  seoTitle        String          @default("")
  seoDescription  String          @default("")
  seoKeywords     String          @default("")
  publishedAt     DateTime?
  createdAt       DateTime        @default(now())
  updatedAt       DateTime        @updatedAt
  tags            ArticleTag[]
  likes           ArticleLike[]
  comments        ArticleComment[]
}

model ArticleCategory {
  id           String    @id @default(cuid())
  slug         String    @unique
  name         String
  description  String    @default("")
  color        String    @default("#f97316")
  icon         String    @default("BookOpen")
  order        Int       @default(0)
  articleCount Int       @default(0)
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt
  articles     Article[]
}

model ArticleTag {
  id        String   @id @default(cuid())
  slug      String   @unique
  name      String
  createdAt DateTime @default(now())
  articles  Article[]
}

model ArticleLike {
  id        String   @id @default(cuid())
  userId    String
  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  articleId String
  article   Article  @relation(fields: [articleId], references: [id], onDelete: Cascade)
  createdAt DateTime @default(now())
  @@unique([userId, articleId])
}

model ArticleComment {
  id          String           @id @default(cuid())
  articleId   String
  article     Article          @relation(fields: [articleId], references: [id], onDelete: Cascade)
  userId      String?
  user        User?            @relation(fields: [userId], references: [id], onDelete: SetNull)
  guestName   String           @default("زائر")
  guestAvatar String?
  parentId    String?
  parent      ArticleComment?  @relation("Replies", fields: [parentId], references: [id], onDelete: Cascade)
  replies     ArticleComment[] @relation("Replies")
  text        String
  likes       Int              @default(0)
  dislikes    Int              @default(0)
  isPinned    Boolean          @default(false)
  isEdited    Boolean          @default(false)
  createdAt   DateTime         @default(now())
  updatedAt   DateTime         @updatedAt
}

// ===== المنتدى =====
model ForumCategory {
  id          String       @id @default(cuid())
  slug        String       @unique
  name        String
  description String       @default("")
  color       String       @default("#3b82f6")
  icon        String       @default("MessageSquare")
  order       Int          @default(0)
  threadCount Int          @default(0)
  postCount   Int          @default(0)
  isLocked    Boolean      @default(false)
  createdAt   DateTime     @default(now())
  updatedAt   DateTime     @updatedAt
  threads     ForumThread[]
}

model ForumThread {
  id            String        @id @default(cuid())
  slug          String        @unique
  title         String
  authorId      String
  author        User          @relation(fields: [authorId], references: [id], onDelete: Cascade)
  categoryId    String
  category      ForumCategory @relation(fields: [categoryId], references: [id], onDelete: Cascade)
  isPinned      Boolean       @default(false)
  isLocked      Boolean       @default(false)
  isSolved      Boolean       @default(false)
  isFeatured    Boolean       @default(false)
  views         Int           @default(0)
  postCount     Int           @default(0)
  lastReplyAt   DateTime?
  lastReplyById String?
  tags          String        @default("")
  createdAt     DateTime      @default(now())
  updatedAt     DateTime      @updatedAt
  posts         ForumPost[]
  threadLikes   ForumThreadLike[]
}

model ForumPost {
  id         String      @id @default(cuid())
  threadId   String
  thread     ForumThread @relation(fields: [threadId], references: [id], onDelete: Cascade)
  authorId   String
  author     User        @relation(fields: [authorId], references: [id], onDelete: Cascade)
  parentId   String?
  parent     ForumPost?  @relation("Replies", fields: [parentId], references: [id], onDelete: Cascade)
  replies    ForumPost[] @relation("Replies")
  content    String      // markdown
  likes      Int         @default(0)
  dislikes   Int         @default(0)
  isEdited   Boolean     @default(false)
  isSolution Boolean     @default(false)
  createdAt  DateTime    @default(now())
  updatedAt  DateTime    @updatedAt
}

model ForumThreadLike {
  id        String      @id @default(cuid())
  userId    String
  user      User        @relation(fields: [userId], references: [id], onDelete: Cascade)
  threadId  String
  thread    ForumThread @relation(fields: [threadId], references: [id], onDelete: Cascade)
  createdAt DateTime    @default(now())
  @@unique([userId, threadId])
}

model ForumPostLike {
  id        String    @id @default(cuid())
  userId    String
  user      User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  postId    String
  post      ForumPost @relation(fields: [postId], references: [id], onDelete: Cascade)
  value     String    @default("like") // like | dislike
  createdAt DateTime  @default(now())
  @@unique([userId, postId])
}
```
